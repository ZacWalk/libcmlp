cmake_minimum_required(VERSION 3.24)
project(cmlp_consumer LANGUAGES C)
include(CTest)
if(NOT LIBCMLP_ROOT)
    message(FATAL_ERROR "LIBCMLP_ROOT is required")
endif()
add_subdirectory("${LIBCMLP_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/cmlp")
if(TARGET mlp-cli OR TARGET cmlp-tests OR TARGET cmlp-pipeline-tests)
    message(FATAL_ERROR "Embedding cmlp must not add standalone examples or repository tests")
endif()
add_executable(consumer main.c)
target_link_libraries(consumer PRIVATE cmlp)
add_test(NAME consumer.smoke COMMAND consumer)
